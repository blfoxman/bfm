RPCs CREATE OR REPLACE FUNCTION verify_login(p_email text, p_password text) RETURNS jsonb LANGUAGE plpgsql SECURITY DEFINER SET search_path = public, extensions AS $$ DECLARE v_user pf_users%ROWTYPE; v_token text; BEGIN SELECT * INTO v_user FROM pf_users WHERE email=p_email AND approved=true; IF v_user.id IS NULL THEN RETURN jsonb_build_object('ok',false,'error','Invalid email or password.'); END IF; END IF; END; $$; CREATE OR REPLACE FUNCTION secure_write(p_token text, p_table text, p_op text, p_id text DEFAULT NULL, p_data jsonb DEFAULT NULL) RETURNS jsonb LANGUAGE plpgsql SECURITY DEFINER SET search_path = public, extensions AS $$ DECLARE v_user_id text;
